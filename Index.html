uint256 public policyCounter; uint256 public totalReserveLiquidity; uint256 public accumulatedFees; bool public isProtocolHalted; mapping(address => bool) public authorizedKeepers; mapping(uint256 => CoveragePolicy) public policies; _; } constructor() { syndicateDirector = payable(msg.sender); protocolDescriptor = "Apex Base On-Chain Risk Syndicate Core v1"; basePremiumRateBps = 250; isProtocolHalted = false; authorizedKeepers[msg.sender] = true; } // 1. addAuthorizedKeeper function addAuthorizedKeeper(address keeper) external onlyDirector { require(keeper != address(0), "Syndicate: Zero address"); require(!authorizedKeepers[keeper], "Syndicate: Already keeper"); p.status = PolicyStatus.Canceled; uint256 refund = p.insuredAmount; p.insuredAmount = 0; totalReserveLiquidity -= refund; (bool sent, ) = p.policyHolder.call{value: refund}(""); require(sent, "Syndicate: Refund failed"); emit PolicyCanceled(policyId, refund); } // 4. claimCoveragePayout function claimCoveragePayout(uint256 policyId, uint256 payoutAmount) external onlyKeeperOrDirector { CoveragePolicy storage p = policies[policyId]; require(policyId > 0 && policyId <= policyCounter, "Syndicate: Invalid ID"); require(p.status == PolicyStatus.Active, "Syndicate: Not active"); require(payoutAmount <= p.insuredAmount && payoutAmount > 0, "Syndicate: Invalid payout"); p.status = PolicyStatus.Settled; p.insuredAmount -= payoutAmount; totalReserveLiquidity -= payoutAmount; (bool sent, ) = p.policyHolder.call{value: payoutAmount}(""); require(sent, "Syndicate: Claim payout failed"); emit CoverageClaimed(policyId, payoutAmount); } // 5. depositReserveLiquidity function depositReserveLiquidity() external payable onlyDirector { require(msg.value > 0, "Syndicate: Zero deposit"); totalReserveLiquidity += msg.value; if (surplus == 0 && address(this).balance > 0 && policyCounter == 0) { surplus = address(this).balance; } require(surplus > 0, "Syndicate: No surplus"); (bool sent, ) = syndicateDirector.call{value: surplus}(""); require(sent, "Syndicate: Drain failed"); emit ReservesDrained(syndicateDirector, surplus); } // 7. extendPolicyExpiry function extendPolicyExpiry(uint256 policyId, uint256 extraSeconds) external whenNotHalted { CoveragePolicy storage p = policies[policyId]; require(policyId > 0 && policyId <= policyCounter, "Syndicate: Invalid ID"); require(p.policyHolder == msg.sender || msg.sender == syndicateDirector, "Syndicate: Forbidden"); require(p.status == PolicyStatus.Active, "Syndicate: Not active"); require(extraSeconds > 0, "Syndicate: Invalid extra time"); p.expiresAt += extraSeconds; emit PolicyExtended(policyId, p.expiresAt); } // 8. harvestProtocolFees function harvestProtocolFees() external onlyDirector { uint256 fees = accumulatedFees; require(fees > 0, "Syndicate: Zero fees"); accumulatedFees = 0; (bool sent, ) = syndicateDirector.call{value: fees}(""); require(sent, "Syndicate: Fee harvest failed"); emit FeesHarvested(syndicateDirector, fees); } // 9. issueInsurancePolicy function issueInsurancePolicy( bytes32 riskCoverTag, uint256 durationSeconds ) external payable whenNotHalted returns (uint256) { require(msg.value > 0, "Syndicate: Premium required"); totalReserveLiquidity += netCover; emit PolicyIssued(policyCounter, msg.sender, netCover); return policyCounter; } // 10. rebalancePoolCapacity function rebalancePoolCapacity(uint256 adjustmentDelta) external onlyDirector { totalReserveLiquidity += adjustmentDelta;
